    sortBar.style.display = '';
    btnL.style.display = ''; btnH.style.display = 'none';
    setSort('latest', btnL);
  } else if (id === 'ghtrend') {
    sortBar.style.display = '';
    btnL.style.display = 'none'; btnH.style.display = '';
    setSort('hotness', btnH);
  } else {
    sortBar.style.display = '';
    btnL.style.display = ''; btnH.style.display = '';

  }
